function doPost(e) { var lock = LockService.getScriptLock(); lock.tryLock(10000); try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = JSON.parse(e.postData.contents); // Cabeçalho da base econométrica if (sheet.getLastRow() === 0) { sheet.appendRow([ "Data/Hora", "ID_Participante", "Grupo_Tratamento", "Sexo", "Ano_Nascimento", "Idade_Aproximada", "Itens_Totais", "Itens_Sustentaveis", "Gasto_Total_RS", "Gasto_Sustentavel_RS", "Percentual_Sustentavel", "Tempo_Decisao_Segundos", "Cliques_Origem_Rastreabilidade", "Lista_Produtos_Carrinho" ]); } // Registro dos dados sheet.appendRow([ data.dataHora, data.idParticipante, data.grupo, data.sexo, data.anoNascimento, data.idadeAproximada, data.itensTotais, data.itensSustentaveis, data.gastoTotal, data.gastoSustentavel, data.percSustentavel, data.tempoSegundos, data.cliquesOrigem, data.produtos ]); return ContentService .createTextOutput(JSON.stringify({ result: "success" })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService .createTextOutput(JSON.stringify({ result: "error", error: error })) .setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } }
